# Exemplo: Consulta à API Shodan
import shodan

API_KEY = "sua_chave_aqui"
api = shodan.Shodan(API_KEY)

def search_ip(ip):
    try:
        # Busca informações do IP
        host = api.host(ip)
        
        print(f"IP: {host['ip_str']}")
        print(f"Organização: {host.get('org', 'N/A')}")
        print(f"Sistema Operacional: {host.get('os', 'N/A')}")
        
        # Portas abertas
        for item in host['data']:
            print(f"Porta {item['port']}: {item.get('product', '')}")
            
    except shodan.APIError as e:
        print(f"Erro: {e}")

# Uso ético apenas
search_ip("8.8.8.8")
// Exemplo: Consulta à API VirusTotal
async function analyzeFile(fileHash) {
    const apiKey = 'sua_chave_aqui';
    const url = `https://www.virustotal.com/api/v3/files/${fileHash}`;
    
    const response = await fetch(url, {
        headers: {
            'x-apikey': apiKey
        }
    });
    
    const data = await response.json();
    
    // Análise dos resultados
    if (data.data.attributes.last_analysis_results) {
        const results = data.data.attributes.last_analysis_results;
        let malicious = 0;
        
        for (const engine in results) {
            if (results[engine].category === 'malicious') {
                malicious++;
            }
        }
        
        console.log(`Arquivo: ${fileHash}`);
        console.log(`Detecções maliciosas: ${malicious}`);
    }
}

// Uso responsável
analyzeFile('hash_do_arquivo');
